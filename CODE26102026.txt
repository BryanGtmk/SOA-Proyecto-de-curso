import logging

from uuid import uuid4

from fastapi.responses import JSONResponse
from fastapi.exceptions import RequestValidationError

from starlette.exceptions import HTTPException

from sqlalchemy.exc import SQLAlchemyError, IntegrityError

class ApiError(Exception):

    def __init__(self, status, codigo, detalle):

        self.status, self.codigo, self.detalle = status, codigo, detalle

def respuesta(req, status, codigo, detalle):

    trace = str(uuid4())

    return JSONResponse(status_code=status,

        media_type="application/problem+json", content={

            "type": "urn:ventas:" + codigo.lower(),

            "title": codigo, "status": status, "detail": detalle,

            "instance": req.url.path, "codigo": codigo, "traceId": trace,
        })

def instalar(app):

    @app.exception_handler(ApiError)

    async def negocio(req, ex):

        return respuesta(req, ex.status, ex.codigo, ex.detalle)

    @app.exception_handler(RequestValidationError)

    async def validar(req, ex):

        return respuesta(req, 422, "VALIDACION", "; ".join(

            ".".join(map(str, e["loc"])) + ": " + e["msg"]

            for e in ex.errors()))

    @app.exception_handler(HTTPException)

    async def http(req, ex):

        return respuesta(req, ex.status_code, "ERROR_HTTP", str(ex.detail))

    @app.exception_handler(IntegrityError)

    async def integridad(req, ex):

        return respuesta(req, 409, "CONFLICTO_DATOS", "Registro duplicado")

    @app.exception_handler(SQLAlchemyError)

    async def bd(req, ex):

        logging.error("Fallo de persistencia", exc_info=(type(ex), ex, ex.__traceback__))

        return respuesta(req, 503, "BD_NO_DISPONIBLE",

                         "No se pudo acceder a la persistencia")

    @app.exception_handler(Exception)

    async def inesperado(req, ex):

        logging.error("Fallo inesperado", exc_info=(type(ex), ex, ex.__traceback__))

        return respuesta(req, 500, "ERROR_INTERNO", "Error interno registrado")

import httpx

from fastapi import Header

from app.config import JAVA_URL
from app.errors import ApiError

def identidad(authorization: str | None = Header(default=None)):

    if not authorization or not authorization.startswith("Bearer "):

        raise ApiError(401, "NO_AUTENTICADO", "Enviar access token")

    try:

        r = httpx.get(JAVA_URL + "/api/identidad",

                      headers={"Authorization": authorization}, timeout=5)

    except httpx.RequestError:

        raise ApiError(503, "IDENTIDAD_NO_DISPONIBLE", "Java no responde")

    if r.status_code in (401, 403):

        raise ApiError(r.status_code, "ACCESO_RECHAZADO", "Token no autorizado")

    if r.status_code != 200:

        raise ApiError(503, "IDENTIDAD_NO_DISPONIBLE", "No se validó la identidad")

    try:

        data = r.json()

        if not isinstance(data, dict):

            raise ValueError("No es un objeto")

        if not isinstance(data.get("usuario"), str):

            raise ValueError("Usuario ausente")

        if not isinstance(data.get("roles"), list):

            raise ValueError("Roles ausentes")

    except (ValueError, TypeError):

        raise ApiError(503, "IDENTIDAD_INVALIDA", "Respuesta de Java inválida")

    if len(data["usuario"]) > 100:

        raise ApiError(403, "IDENTIDAD_INVALIDA", "Usuario demasiado largo")

    return data

def vendedor(usuario):

    if not {"ROLE_OPERADOR", "ROLE_ADMIN"}.intersection(usuario["roles"]):

        raise ApiError(403, "SIN_PERMISO", "Se requiere OPERADOR o ADMIN")

def propietario(usuario, dueno):

    if usuario["usuario"] != dueno and "ROLE_ADMIN" not in usuario["roles"]:

        raise ApiError(403, "SIN_PERMISO", "Recurso de otro usuario")

import httpx

from app.config import JAVA_URL, SERVICE_KEY

class FalloRemoto(Exception):

    def __init__(self, status, detalle):

        self.status, self.detalle = status, detalle

class VehiculosClient:

    def llamar(self, metodo, ruta, **kwargs):

        try:

            r = httpx.request(metodo, JAVA_URL + ruta,

                headers={"X-Service-Key": SERVICE_KEY},

                timeout=httpx.Timeout(5, connect=2), **kwargs)

        except httpx.RequestError:

            raise FalloRemoto(503, "Resultado remoto desconocido")

        if r.status_code >= 400:

            raise FalloRemoto(r.status_code, "Operación Java rechazada")

        try:

            return r.json() if r.content else None

        except ValueError:

            raise FalloRemoto(502, "Respuesta Java ilegible")

    def por_operacion(self, op):

        return self.llamar("GET", "/api/reservas",

                           params={"operacionId": op})

    def reservar(self, vehiculo, op):

        return self.llamar("POST", f"/api/vehiculos/{vehiculo}/reservas",

                           json={"operacionId": op})

    def confirmar(self, reserva):

        return self.llamar("PUT", f"/api/reservas/{reserva}/confirmacion")

from sqlalchemy import select, or_

from app.models import Venta

class VentaRepository:

    def __init__(self, db):

        self.db = db

    def por_clave(self, usuario, clave):

        return self.db.scalar(select(Venta).where(

            Venta.usuario == usuario, Venta.idempotency_key == clave))

    def bloquear(self, id):

        return self.db.scalar(select(Venta).where(Venta.id == id)

                              .with_for_update())

    def pendientes(self):

        return self.db.scalars(select(Venta.id).where(or_(

            Venta.estado.in_(["PENDIENTE", "EN_REVISION"]),

            (Venta.estado == "COMPLETADA") &

            (Venta.notificacion != "ENVIADA")))

            .order_by(Venta.id).limit(100)).all()

import hashlib

import json

from uuid import uuid4

from decimal import Decimal

import httpx

from sqlalchemy.exc import IntegrityError

from app.db import SessionLocal
from app.models import Venta, Comprador

from app.repositories.venta_repository import VentaRepository
from app.clients.vehiculos_client import VehiculosClient, FalloRemoto

from app.config import NOTIF_URL, NOTIF_KEY
from app.errors import ApiError
from app.security import propietario

remote = VehiculosClient()

def salida(v):

    return {"id": v.id, "operacionId": v.operacion_id,

            "vehiculoId": v.vehiculo_id, "compradorId": v.comprador_id,

            "precioAcordado": str(v.precio_acordado)

                if v.precio_acordado is not None else None,

            "moneda": v.moneda, "estado": v.estado,

            "notificacion": v.notificacion,

            "fechaUTC": v.fecha.isoformat() + "Z"}

def crear(datos, clave, usuario):

    cuerpo = json.dumps(datos.model_dump(), sort_keys=True,

                        separators=(",", ":"))

    huella = hashlib.sha256(cuerpo.encode()).hexdigest()

    nueva = False

    with SessionLocal() as db:

        repo = VentaRepository(db)

        v = repo.por_clave(usuario["usuario"], clave)

        if v is None:

            comprador = db.get(Comprador, datos.compradorId)

            if comprador is None:

                raise ApiError(404, "COMPRADOR_NO_EXISTE", "Comprador inexistente")

            propietario(usuario, comprador.usuario)

            v = Venta(operacion_id=str(uuid4()), usuario=usuario["usuario"],

                      vehiculo_id=datos.vehiculoId, comprador_id=datos.compradorId,

                      idempotency_key=clave, request_hash=huella)

            db.add(v)

            try:

                db.commit()

                nueva = True

            except IntegrityError:

                db.rollback()

                v = repo.por_clave(usuario["usuario"], clave)

                if v is None:

                    raise

        if v.request_hash != huella:

            raise ApiError(409, "CLAVE_REUTILIZADA",

                           "La misma clave requiere el mismo cuerpo")

        id = v.id

    procesar(id)

    notificar(id)

    with SessionLocal() as db:

        v = db.get(Venta, id)

        if v.estado == "FALLIDA":

            raise ApiError(v.error_status or 409, "VENTA_RECHAZADA",

                           v.error_detalle or "La venta fue rechazada")

        status = (201 if nueva else 200) if v.estado == "COMPLETADA" else 202

        return salida(v), status

def procesar(id):

    with SessionLocal.begin() as db:

        v = VentaRepository(db).bloquear(id)

        if v is None or v.estado in ("COMPLETADA", "FALLIDA"):

            return

        try:

            try:

                r = remote.por_operacion(v.operacion_id)

            except FalloRemoto as ex:

                if ex.status != 404:

                    raise

                try:

                    r = remote.reservar(v.vehiculo_id, v.operacion_id)

                except FalloRemoto as rechazo:

                    if rechazo.status in (404, 409, 422):

                        v.estado = "FALLIDA"

                        v.error_status = rechazo.status

                        v.error_detalle = "Vehículo inexistente o no disponible"

                        return

                    raise

            v.reserva_id = r["id"]

            v.precio_acordado = Decimal(str(r["precio"]))

            v.moneda = r["moneda"]

            if r["estado"] == "LIBERADA":

                v.estado = "FALLIDA"; v.error_status = 409

                v.error_detalle = "Reserva liberada"

                return

            if r["estado"] == "RESERVADA":

                r = remote.confirmar(r["id"])

            if r["estado"] != "CONFIRMADA":

                raise FalloRemoto(502, "Estado de reserva inesperado")

            v.estado = "COMPLETADA"

            v.error_status = None; v.error_detalle = None

        except (FalloRemoto, ValueError, KeyError) as ex:

            v.estado = "EN_REVISION"

            v.error_detalle = "Pendiente de consultar y conciliar con Java"

def notificar(id):

    with SessionLocal.begin() as db:

        v = VentaRepository(db).bloquear(id)

        if v is None or v.estado != "COMPLETADA" or v.notificacion == "ENVIADA":

            return

        try:

            r = httpx.post(NOTIF_URL + "/api/notificaciones", timeout=3,

                headers={"X-Notification-Key": NOTIF_KEY}, json={

                    "vehiculoId": v.vehiculo_id, "tipo": "VENTA",

                    "destinatario": v.comprador.correo,

                    "mensaje": f"Venta {v.id} operación {v.operacion_id} completada",
                })

            v.notificacion = "ENVIADA" if 200 <= r.status_code < 300 else "PENDIENTE"

        except httpx.RequestError:

            v.notificacion = "PENDIENTE"

from typing import Annotated

from fastapi import APIRouter, Depends, Header, Response

from app.db import SessionLocal
from app.models import Venta, Comprador
from app.schemas import CrearVenta, CrearComprador
from app.security import identidad, vendedor, propietario
from app.errors import ApiError
from app.services import venta_service as service

router = APIRouter(prefix="/api")

@router.post("/compradores", status_code=201)

def crear_comprador(d: CrearComprador, u=Depends(identidad)):

    vendedor(u)

    with SessionLocal.begin() as db:

        c = Comprador(usuario=u["usuario"], **d.model_dump())

        db.add(c); db.flush()

        return {"id": c.id, "nombre": c.nombre, "correo": c.correo}

@router.post("/ventas")

def crear(d: CrearVenta, response: Response,

          idempotency_key: Annotated[str, Header(

              min_length=8, max_length=100, pattern=r"^[A-Za-z0-9_-]+$")],

          u=Depends(identidad)):

    vendedor(u)

    resultado, codigo = service.crear(d, idempotency_key, u)

    response.status_code = codigo

    response.headers["Location"] = f"/api/ventas/{resultado['id']}"

    return resultado

@router.get("/ventas/{id}")

def obtener(id: int, u=Depends(identidad)):

    with SessionLocal() as db:

        v = db.get(Venta, id)

        if v is None:

            raise ApiError(404, "VENTA_NO_EXISTE", "Venta inexistente")

        propietario(u, v.usuario)

        return service.salida(v)

import sys

import time

import logging

from app.db import SessionLocal

from app.repositories.venta_repository import VentaRepository

from app.services.venta_service import procesar, notificar

def ciclo():

    with SessionLocal() as db:

        ids = VentaRepository(db).pendientes()

    for id in ids:

        try:

            procesar(id)

            notificar(id)

        except Exception:

            logging.exception("No se pudo conciliar venta %s", id)

if __name__ == "__main__":

    while True:

        try:

            ciclo()

        except Exception:

            logging.exception("Fallo de ciclo de recuperación")

        if "--once" in sys.argv:

            break

        time.sleep(10)

from alembic import context

from app.db import Base, engine

from app import models

TABLAS_PYTHON = {"compradores", "ventas"}

target_metadata = Base.metadata

def incluir_nombre(nombre, tipo, nombres_padre):

    if tipo == "table":

        return nombre in TABLAS_PYTHON

    return True

def incluir_objeto(objeto, nombre, tipo, reflejado, comparar):

    if tipo == "table":

        return nombre in TABLAS_PYTHON

    return True

def run_migrations_offline():

    context.configure(

        url=str(engine.url),

        target_metadata=target_metadata,

        include_name=incluir_nombre,

        include_object=incluir_objeto,

        compare_type=True,

        literal_binds=True,
    )

    with context.begin_transaction():

        context.run_migrations()

def run_migrations_online():

    with engine.connect() as connection:

        context.configure(

            connection=connection,

            target_metadata=target_metadata,

            include_name=incluir_nombre,

            include_object=incluir_objeto,

            compare_type=True,
        )

        with context.begin_transaction():

            context.run_migrations()

if context.is_offline_mode():

    run_migrations_offline()

else:

    run_migrations_online()
