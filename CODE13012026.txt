package ec.edu.uta.venta_vehiculos_api.exception;

import java.net.URI;

import java.util.UUID;

import org.slf4j.LoggerFactory;

import org.springframework.dao.DataIntegrityViolationException;

import org.springframework.http.*;

import org.springframework.http.converter.HttpMessageNotReadableException;

import org.springframework.web.bind.MethodArgumentNotValidException;

import org.springframework.web.bind.annotation.*;

import org.springframework.web.ErrorResponse;

import jakarta.servlet.http.HttpServletRequest;

@RestControllerAdvice
public class ApiExceptionHandler {

    @ExceptionHandler(Exception.class)

    public ResponseEntity<ProblemDetail> manejar(
            Exception ex, HttpServletRequest req) {

        int status = 500;

        String codigo = "ERROR_INTERNO";

        String detalle = "No fue posible completar la solicitud";

        if (ex instanceof ApiError e) {

            status = e.status;
            codigo = e.codigo;
            detalle = e.getMessage();

        } else if (ex instanceof MethodArgumentNotValidException e) {

            status = 422;
            codigo = "VALIDACION";

            detalle = e.getBindingResult().getFieldErrors().stream()

                .map(x -> x.getField() + ": " + x.getDefaultMessage())

                .reduce((a,b) -> a + "; " + b)

                .orElse("Datos inválidos");

        } else if (ex instanceof HttpMessageNotReadableException) {

            status = 400;
            codigo = "JSON_INVALIDO";
            detalle = "JSON ilegible o tipo de dato incorrecto";

        } else if (ex instanceof DataIntegrityViolationException) {

            status = 409;
            codigo = "CONFLICTO_DATOS";
            detalle = "Duplicado o relación que impide la operación";

        } else if (ex instanceof IllegalArgumentException) {

            status = 400;
            codigo = "SOLICITUD_INVALIDA";
            detalle = ex.getMessage();

        } else if (ex instanceof ErrorResponse e) {

            status = e.getStatusCode().value();

            codigo = "ERROR_HTTP";

            detalle = e.getBody().getDetail();
        }

        String trace = UUID.randomUUID().toString();

        if (status >= 500)
            LoggerFactory.getLogger(getClass())
                .error("Error " + trace, ex);

        ProblemDetail p = ProblemDetail.forStatusAndDetail(
            HttpStatusCode.valueOf(status), detalle);

        p.setType(URI.create("urn:vehiculos:" + codigo.toLowerCase()));

        p.setInstance(URI.create(req.getRequestURI()));

        p.setProperty("codigo", codigo);

        p.setProperty("traceId", trace);

        return ResponseEntity.status(status)
            .contentType(MediaType.APPLICATION_PROBLEM_JSON)
            .body(p);

    }

}

package ec.edu.uta.venta_vehiculos_api.service;

import java.math.BigDecimal;

import java.time.Year;

import java.util.*;

import org.springframework.stereotype.Service;

import org.springframework.transaction.annotation.Transactional;

import ec.edu.uta.venta_vehiculos_api.dto.*;

import ec.edu.uta.venta_vehiculos_api.exception.ApiError;

import ec.edu.uta.venta_vehiculos_api.model.Vehiculo;

import ec.edu.uta.venta_vehiculos_api.repository.*;

@Service

@Transactional
public class VehiculoService {

    private final VehiculoRepository repo;

    private final ReservaRepository reservas;

    public VehiculoService(
            VehiculoRepository repo,
            ReservaRepository reservas) {

        this.repo = repo;

        this.reservas = reservas;
    }

    @Transactional(readOnly = true)
    public List<VehiculoResponse> listar(String marca) {

        var lista = marca == null
                ? repo.findAll()
                : repo.findByMarcaContainingIgnoreCase(marca);

        return lista.stream()
                .map(VehiculoResponse::de)
                .toList();
    }

    @Transactional(readOnly = true)
    public VehiculoResponse obtener(int id) {

        return VehiculoResponse.de(
                repo.findById(id)
                    .orElseThrow(() -> falta())
        );
    }

    public VehiculoResponse crear(VehiculoRequest d) {

        Vehiculo v = new Vehiculo();

        copiar(v, d);

        v.setEstado("DISPONIBLE");

        return VehiculoResponse.de(repo.saveAndFlush(v));
    }

    public VehiculoResponse actualizar(int id, VehiculoRequest d) {

        Vehiculo v = editable(id);

        copiar(v, d);

        return VehiculoResponse.de(repo.saveAndFlush(v));
    }

    public VehiculoResponse precio(int id, Map<String, Object> d) {

        if (d.size() != 1 ||
                !d.containsKey("precio") ||
                !(d.get("precio") instanceof Number)) {

            throw new ApiError(
                    422,
                    "PATCH_INVALIDO",
                    "PATCH admite únicamente precio numérico y no nulo"
            );
        }

        BigDecimal precio;

        try {

            precio = new BigDecimal(d.get("precio").toString());

        } catch (NumberFormatException e) {

            throw new ApiError(
                    422,
                    "PRECIO_INVALIDO",
                    "Precio inválido"
            );
        }

        if (precio.signum() <= 0 ||
                precio.scale() > 2 ||
                precio.compareTo(
                    new BigDecimal("9999999999.99")) > 0) {

            throw new ApiError(
                    422,
                    "PRECIO_INVALIDO",
                    "Precio positivo con máximo dos decimales"
            );
        }

        Vehiculo v = editable(id);

        v.setPrecio(precio);

        return VehiculoResponse.de(repo.saveAndFlush(v));
    }

    public void eliminar(int id) {

        Vehiculo v = editable(id);

        if (reservas.existsByVehiculo_IdVehiculo(id)) {

            throw new ApiError(
                    409,
                    "HISTORIAL_RESERVAS",
                    "Se conserva el vehículo por su historial de reservas"
            );
        }

        repo.delete(v);

        repo.flush();
    }

    private Vehiculo editable(int id) {

        Vehiculo v = repo.bloquear(id)
                .orElseThrow(() -> falta());

        if (!"DISPONIBLE".equals(v.getEstado())) {

            throw new ApiError(
                    409,
                    "VEHICULO_NO_DISPONIBLE",
                    "Un vehículo reservado o vendido no se modifica ni elimina"
            );
        }

        return v;
    }

    private ApiError falta() {

        return new ApiError(
                404,
                "VEHICULO_NO_ENCONTRADO",
                "El vehículo no existe"
        );
    }

    private void copiar(Vehiculo v, VehiculoRequest d) {

        if (d.anio() < 1900 ||
                d.anio() > Year.now().getValue() + 1) {

            throw new ApiError(
                    422,
                    "ANIO_INVALIDO",
                    "Año fuera de rango"
            );
        }

        v.setMarca(d.marca().trim());

        v.setModelo(d.modelo().trim());

        v.setPlaca(
                d.placa().trim().toUpperCase(Locale.ROOT)
        );

        v.setChasis(
                d.chasis().trim().toUpperCase(Locale.ROOT)
        );

        if (v.getChasis().length() != 17) {

            throw new ApiError(
                    422,
                    "CHASIS_INVALIDO",
                    "Chasis de 17 caracteres"
            );
        }

        v.setAnio(d.anio());

        v.setColor(d.color());

        v.setPrecio(d.precio());
    }

}

package ec.edu.uta.venta_vehiculos_api.service;

import java.util.UUID;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.*;
import ec.edu.uta.venta_vehiculos_api.dto.*;
import ec.edu.uta.venta_vehiculos_api.exception.ApiError;
import ec.edu.uta.venta_vehiculos_api.model.*;
import ec.edu.uta.venta_vehiculos_api.repository.*;

@Service

@Transactional(isolation=Isolation.READ_COMMITTED)
public class ReservaService {
 private final VehiculoRepository vehiculos;
 private final ReservaRepository reservas;

 public ReservaService(VehiculoRepository v, ReservaRepository r) {
  vehiculos=v; reservas=r;
 }

 public ReservaResponse reservar(int id, String op) {

  Vehiculo v=vehiculos.bloquear(id).orElseThrow(() ->
   new ApiError(404,"VEHICULO_NO_ENCONTRADO","Vehículo inexistente"));

  var anterior=reservas.findByOperacionId(op);

  if (anterior.isPresent()) {
   Reserva r=anterior.get();

   if (!r.vehiculo.getIdVehiculo().equals(id))

    throw conflicto("La operación pertenece a otro vehículo");

   return ReservaResponse.de(r);
  }

  if (!"DISPONIBLE".equals(v.getEstado()))

   throw conflicto("El vehículo no está disponible");
  Reserva r=new Reserva();
  r.id=UUID.randomUUID().toString(); r.operacionId=op;
  r.vehiculo=v; r.precio=v.getPrecio();
  r.moneda="USD"; r.estado="RESERVADA";
  v.setEstado("RESERVADO");
  reservas.saveAndFlush(r);

  return ReservaResponse.de(r);
 }

 @Transactional(readOnly=true)

 public ReservaResponse obtener(String id) {

  return ReservaResponse.de(buscar(id));
 }

 @Transactional(readOnly=true)

 public ReservaResponse porOperacion(String op) {

  return ReservaResponse.de(reservas.findByOperacionId(op)
   .orElseThrow(() -> new ApiError(404,"RESERVA_NO_ENCONTRADA",
    "La operación no tiene reserva")));
 }

 public ReservaResponse confirmar(String id) {
  Reserva r=bloquear(id);

  if ("LIBERADA".equals(r.estado))

   throw conflicto("La reserva fue liberada");

  if ("RESERVADA".equals(r.estado)) {

   if (!"RESERVADO".equals(r.vehiculo.getEstado()))

    throw conflicto("Estado del vehículo inconsistente");
   r.estado="CONFIRMADA"; r.vehiculo.setEstado("VENDIDO");
  }

  return ReservaResponse.de(r);
 }

 public void liberar(String id) {
  Reserva r=bloquear(id);

  if ("CONFIRMADA".equals(r.estado))

   throw conflicto("No se libera una reserva confirmada");

  if ("RESERVADA".equals(r.estado)) {
   r.estado="LIBERADA"; r.vehiculo.setEstado("DISPONIBLE");
  }
 }

 private Reserva bloquear(String id) {
  Reserva r=buscar(id);

  vehiculos.bloquear(r.vehiculo.getIdVehiculo()).orElseThrow();

  return refrescar(r);
 }
 @jakarta.persistence.PersistenceContext
 private jakarta.persistence.EntityManager em;

 private Reserva refrescar(Reserva r) {
  em.refresh(r); em.refresh(r.vehiculo); return r;
 }

 private Reserva buscar(String id) {

  return reservas.findById(id).orElseThrow(() ->
   new ApiError(404,"RESERVA_NO_ENCONTRADA","Reserva inexistente"));
 }

 private ApiError conflicto(String detalle) {

  return new ApiError(409,"CONFLICTO_RESERVA",detalle);
 }
}

import java.util.List;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;
import jakarta.servlet.*;
import jakarta.servlet.http.*;

@Component
public class IntegrationFilter extends OncePerRequestFilter {
 private final String key;

 public IntegrationFilter(@Value("${app.security.ventas-key}") String key) {

  if (key.length()<32) throw new IllegalStateException("Clave de servicio corta");
  this.key=key;
 }
 @Override
 protected void doFilterInternal(HttpServletRequest req,
   HttpServletResponse res, FilterChain chain)
   throws IOException, ServletException {
  String path=req.getRequestURI();
  boolean interna=path.equals("/api/reservas") ||
   path.startsWith("/api/reservas/") ||
   path.matches("/api/vehiculos/[0-9]+/reservas");

  if (interna) {

   String enviada=req.getHeader("X-Service-Key");

   if (enviada==null || !MessageDigest.isEqual(
    key.getBytes(StandardCharsets.UTF_8),
    enviada.getBytes(StandardCharsets.UTF_8))) {
    res.setStatus(401); res.setContentType("application/problem+json");
    res.getWriter().write("{\"title\":\"Servicio no autorizado\",\"status\":401}");

    return;
   }

   SecurityContextHolder.getContext().setAuthentication(
    new UsernamePasswordAuthenticationToken("ventas-api",null,
     List.of(new SimpleGrantedAuthority("ROLE_SERVICIO_VENTAS"))));
  }
  chain.doFilter(req,res);
 }
}

package ec.edu.uta.venta_vehiculos_api.security;

import java.io.IOException;
import java.util.List;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;
import jakarta.servlet.*;
import jakarta.servlet.http.*;

@Component
public class JwtAuthenticationFilter extends OncePerRequestFilter {
 private final JwtService jwt;

 public JwtAuthenticationFilter(JwtService jwt) { this.jwt=jwt; }
 @Override
 protected void doFilterInternal(HttpServletRequest req,
   HttpServletResponse res, FilterChain chain)
   throws IOException, ServletException {

  String h=req.getHeader("Authorization");

  if (SecurityContextHolder.getContext().getAuthentication()==null &&
      h!=null && h.startsWith("Bearer ")) {

   try {
    var c=jwt.validarToken(h.substring(7));

    if (!"access".equals(c.get("tipo",String.class)))

     throw new IllegalArgumentException("Se requiere access token");
    String rol=c.get("rol",String.class);

    if (!List.of("CONSULTA","OPERADOR","ADMIN").contains(rol))

     throw new IllegalArgumentException("Rol inválido");

    SecurityContextHolder.getContext().setAuthentication(
     new UsernamePasswordAuthenticationToken(c.getSubject(),null,
      List.of(new SimpleGrantedAuthority("ROLE_"+rol))));
   } catch (Exception ex) { SecurityContextHolder.clearContext(); }
  }
  chain.doFilter(req,res);
 }
}

package ec.edu.uta.venta_vehiculos_api.security;

import org.springframework.context.annotation.*;
import org.springframework.http.HttpMethod;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.authentication.UsernamePasswordAuthenticationFilter;

@Configuration
public class SecurityConfig {

 @Bean
 public org.springframework.boot.web.servlet.FilterRegistrationBean<JwtAuthenticationFilter>
 desactivarJwtServlet(JwtAuthenticationFilter f) {
  var b=new org.springframework.boot.web.servlet.FilterRegistrationBean<>(f);
  b.setEnabled(false); return b;
 }

 @Bean
 public org.springframework.boot.web.servlet.FilterRegistrationBean<IntegrationFilter>
 desactivarIntegracionServlet(IntegrationFilter f) {
  var b=new org.springframework.boot.web.servlet.FilterRegistrationBean<>(f);
  b.setEnabled(false); return b;
 }

 @Bean public PasswordEncoder passwordEncoder() {

  return new BCryptPasswordEncoder();
 }

 @Bean

 public SecurityFilterChain securityFilterChain(HttpSecurity http,
   JwtAuthenticationFilter jwt, IntegrationFilter integration)
   throws Exception {

  return http.csrf(c -> c.disable())
   .sessionManagement(s -> s.sessionCreationPolicy(
    SessionCreationPolicy.STATELESS))
   .exceptionHandling(e -> e
    .authenticationEntryPoint((q,r,x) -> {
     r.setStatus(401); r.setContentType("application/problem+json");
     r.getWriter().write("{\"title\":\"No autenticado\",\"status\":401}");
    })
    .accessDeniedHandler((q,r,x) -> {
     r.setStatus(403); r.setContentType("application/problem+json");
     r.getWriter().write("{\"title\":\"Sin permiso\",\"status\":403}");
    }))
   .authorizeHttpRequests(a -> a
    .requestMatchers("/oauth/token").permitAll()
    .requestMatchers("/api/reservas", "/api/reservas/**",
     "/api/vehiculos/*/reservas").hasRole("SERVICIO_VENTAS")
    .requestMatchers("/api/identidad")
     .hasAnyRole("CONSULTA","OPERADOR","ADMIN")
    .requestMatchers(HttpMethod.GET,"/api/vehiculos/**")
     .hasAnyRole("CONSULTA","OPERADOR","ADMIN")
    .requestMatchers(HttpMethod.POST,"/api/vehiculos/**")
     .hasAnyRole("OPERADOR","ADMIN")
    .requestMatchers(HttpMethod.PUT,"/api/vehiculos/**")
     .hasAnyRole("OPERADOR","ADMIN")
    .requestMatchers(HttpMethod.PATCH,"/api/vehiculos/**")
     .hasAnyRole("OPERADOR","ADMIN")
    .requestMatchers(HttpMethod.DELETE,"/api/vehiculos/**")
     .hasRole("ADMIN")
    .anyRequest().denyAll())
   .addFilterBefore(jwt,UsernamePasswordAuthenticationFilter.class)
   .addFilterBefore(integration,JwtAuthenticationFilter.class)
   .build();
 }
}

package ec.edu.uta.venta_vehiculos_api.security;

import org.springframework.boot.CommandLineRunner;
import org.springframework.context.annotation.*;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.crypto.password.PasswordEncoder;
import ec.edu.uta.venta_vehiculos_api.repository.UsuarioApiRepository;
import ec.edu.uta.venta_vehiculos_api.model.UsuarioApi;

@Configuration

@Profile("local")
public class LocalUsers {

 @Bean CommandLineRunner usuarioInicial(UsuarioApiRepository repo,
  PasswordEncoder encoder,
  @Value("${app.bootstrap.user}") String user,
  @Value("${app.bootstrap.password}") String password,
  @Value("${app.bootstrap.role:ADMIN}") String rol) {

  return args -> {

   if (!java.util.List.of("ADMIN","OPERADOR","CONSULTA").contains(rol))

    throw new IllegalArgumentException("Rol inicial inválido");

   if (repo.findByUsername(user).isEmpty()) {
    UsuarioApi u=new UsuarioApi(); u.setUsername(user);
    u.setPassword(encoder.encode(password)); u.setRol(rol);

    repo.save(u);
   }
  };
 }
}

package ec.edu.uta.venta_vehiculos_api.client;

import java.time.Duration;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;
import org.springframework.http.client.SimpleClientHttpRequestFactory;
import org.springframework.web.client.*;
import ec.edu.uta.venta_vehiculos_api.dto.NotificacionRequest;

@Component
public class NotificacionClient {
 private final RestClient client;
 private final String destino;

 public NotificacionClient(
  @Value("${servicios.notificaciones.url}") String url,
  @Value("${servicios.notificaciones.key}") String key,
  @Value("${servicios.notificaciones.destinatario}") String destino) {
  var f=new SimpleClientHttpRequestFactory();
  f.setConnectTimeout(Duration.ofSeconds(2));
  f.setReadTimeout(Duration.ofSeconds(3));

  client=RestClient.builder().baseUrl(url).requestFactory(f)

   .defaultHeader("X-Notification-Key",key).build();
  this.destino=destino;
 }

 public void enviar(Integer id,String tipo,String mensaje) {

  try {
   client.post().uri("/api/notificaciones")
    .body(new NotificacionRequest(id,destino,tipo,mensaje))
    .retrieve().toBodilessEntity();
  } catch (RestClientException e) {
   System.err.println("Notificación no enviada para vehículo "+id);
  }
}
}
